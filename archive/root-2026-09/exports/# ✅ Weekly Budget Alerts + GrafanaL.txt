```python
"""
💰 Claude Budget Alerts — Weekly Limits, Thresholds & Notifications
Path: app/claude_budget_alerts.py
Features: Weekly budget · Alert thresholds · Slack/Email alerts · Auto-throttle
"""

import os
from datetime import datetime, timedelta
from typing import Dict, List, Optional, Callable
from dataclasses import dataclass
import json

@dataclass
class BudgetConfig:
    weekly_limit_usd: float = 5.00          # Default: $5/week
    alert_pct: List[float] = None           # Alert at % of budget
    throttle_pct: float = 95.0              # Throttle at 95%
    reset_day: int = 0                       # Reset weekly (0=Monday)

    def __post_init__(self):
        if self.alert_pct is None:
            self.alert_pct = [50.0, 75.0, 90.0]


class BudgetManager:
    def __init__(self, config: BudgetConfig = None, log_path: str = "./logs/budget-state.json"):
        self.cfg = config or BudgetConfig()
        self.log_path = log_path
        self.state = self._load_state()
        self._ensure_storage()

    def _ensure_storage(self):
        os.makedirs(os.path.dirname(self.log_path), exist_ok=True)

    def _load_state(self) -> Dict:
        if os.path.exists(self.log_path):
            with open(self.log_path) as f:
                return json.load(f)
        return self._new_week_state()

    def _new_week_state(self) -> Dict:
        """Calculate current week boundary"""
        now = datetime.utcnow()
        start = now - timedelta(days=now.weekday() - self.cfg.reset_day)
        start = start.replace(hour=0, minute=0, second=0, microsecond=0)
        return {
            "week_start": start.isoformat(),
            "total_spent_usd": 0.0,
            "alerts_sent_pct": [],
            "throttled": False,
        }

    def _check_week_reset(self):
        """Auto-reset if new week started"""
        week_start = datetime.fromisoformat(self.state["week_start"])
        if datetime.utcnow() - week_start >= timedelta(days=7):
            self.state = self._new_week_state()
            self._save_state()

    def record_spend(self, cost_usd: float) -> Dict:
        """Add cost → check thresholds → trigger alerts"""
        self._check_week_reset()
        self.state["total_spent_usd"] += cost_usd
        pct_used = (self.state["total_spent_usd"] / self.cfg.weekly_limit_usd) * 100

        triggered = []
        for threshold in sorted(self.cfg.alert_pct):
            if pct_used >= threshold and threshold not in self.state["alerts_sent_pct"]:
                triggered.append({"pct": threshold, "spent": self.state["total_spent_usd"]})
                self.state["alerts_sent_pct"].append(threshold)

        # Auto-throttle
        if pct_used >= self.cfg.throttle_pct and not self.state["throttled"]:
            self.state["throttled"] = True
            triggered.append({"pct": self.cfg.throttle_pct, "action": "THROTTLE_ACTIVE"})

        self._save_state()
        return {
            "week_start": self.state["week_start"],
            "spent_usd": round(self.state["total_spent_usd"], 4),
            "limit_usd": self.cfg.weekly_limit_usd,
            "pct_used": round(pct_used, 1),
            "remaining_usd": round(self.cfg.weekly_limit_usd - self.state["total_spent_usd"], 4),
            "alerts_triggered": triggered,
            "throttled": self.state["throttled"],
        }

    def is_throttled(self) -> bool:
        self._check_week_reset()
        return self.state["throttled"]

    def get_status(self) -> Dict:
        self._check_week_reset()
        return {
            "week_start": self.state["week_start"],
            "spent_usd": round(self.state["total_spent_usd"], 4),
            "limit_usd": self.cfg.weekly_limit_usd,
            "pct_used": round((self.state["total_spent_usd"] / self.cfg.weekly_limit_usd) * 100, 1),
            "throttled": self.state["throttled"],
        }

    def _save_state(self):
        with open(self.log_path, "w") as f:
            json.dump(self.state, f, indent=2)


# Global Singleton — use anywhere
budget = BudgetManager()
```
