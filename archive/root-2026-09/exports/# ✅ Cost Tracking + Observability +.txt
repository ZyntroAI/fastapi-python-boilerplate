```python
"""
📊 Claude API Observability — Cost, Cache & Usage Tracking
Path: app/claude_observability.py
Purpose: Monitor spending, cache ROI, token usage, and latency
"""

import os
import json
from datetime import datetime
from typing import Dict, List, Optional, Any
from dataclasses import dataclass, asdict

@dataclass
class ModelPricing:
    input_per_million: float       # USD per 1M input tokens
    output_per_million: float      # USD per 1M output tokens
    cache_read_discount: float = 0.90  # 90% off for cache hits

# 2026-09-08 Pricing — USD per 1M tokens
MODEL_PRICING = {
    "claude-opus-4-6": ModelPricing(15.00, 75.00),
    "claude-sonnet-4-6": ModelPricing(3.00, 15.00),
    "claude-haiku-4-5": ModelPricing(0.25, 1.25),
}

@dataclass
class UsageRecord:
    timestamp: str
    model: str
    input_tokens: int
    cache_read_tokens: int
    cache_created_tokens: int
    output_tokens: int
    latency_ms: int
    cost_usd: float
    cache_savings_usd: float
    cache_hit_ratio: float

class ClaudeObservability:
    def __init__(self, storage_path: str = "./logs/claude-usage.jsonl"):
        self.storage_path = storage_path
        self._ensure_storage()

    def _ensure_storage(self):
        os.makedirs(os.path.dirname(self.storage_path), exist_ok=True)
        if not os.path.exists(self.storage_path):
            open(self.storage_path, "w").close()

    def calculate_cost(
        self,
        model: str,
        input_tokens: int,
        cache_read_tokens: int,
        output_tokens: int
    ) -> Dict[str, float]:
        """Calculate actual cost + savings from caching"""
        price = MODEL_PRICING.get(model, MODEL_PRICING["claude-sonnet-4-6"])

        # Cache read tokens = ~10% of normal input cost
        normal_input_cost = (input_tokens / 1_000_000) * price.input_per_million
        cache_read_cost = (cache_read_tokens / 1_000_000) * price.input_per_million * (1 - price.cache_read_discount)
        output_cost = (output_tokens / 1_000_000) * price.output_per_million

        total_cost = normal_input_cost + cache_read_cost + output_cost
        savings_usd = (cache_read_tokens / 1_000_000) * price.input_per_million * price.cache_read_discount
        all_input = input_tokens + cache_read_tokens
        hit_ratio = cache_read_tokens / all_input if all_input > 0 else 0.0

        return {
            "total_usd": round(total_cost, 6),
            "savings_usd": round(savings_usd, 6),
            "cache_hit_ratio": round(hit_ratio, 4),
            "breakdown": {
                "normal_input_usd": round(normal_input_cost, 6),
                "cache_read_usd": round(cache_read_cost, 6),
                "output_usd": round(output_cost, 6),
            }
        }

    def record(
        self,
        model: str,
        usage: Dict[str, int],
        latency_ms: int = 0
    ) -> Dict[str, Any]:
        """Log API call and return cost summary"""
        input_tokens = usage.get("input_tokens", 0)
        cache_read = usage.get("cache_read_input_tokens", 0)
        cache_created = usage.get("cache_creation_input_tokens", 0)
        output_tokens = usage.get("output_tokens", 0)

        cost = self.calculate_cost(model, input_tokens, cache_read, output_tokens)

        record = UsageRecord(
            timestamp=datetime.utcnow().isoformat(),
            model=model,
            input_tokens=input_tokens,
            cache_read_tokens=cache_read,
            cache_created_tokens=cache_created,
            output_tokens=output_tokens,
            latency_ms=latency_ms,
            cost_usd=cost["total_usd"],
            cache_savings_usd=cost["savings_usd"],
            cache_hit_ratio=cost["cache_hit_ratio"],
        )

        self._append(asdict(record))
        return {"record": asdict(record), "cost": cost}

    def _append(self, data: dict):
        with open(self.storage_path, "a") as f:
            f.write(json.dumps(data) + "\n")

    def get_summary(self, days: int = 7) -> Dict[str, Any]:
        """Rollup summary for recent usage"""
        records = self._load_recent(days)
        if not records:
            return {"period_days": days, "calls": 0}

        total_input = sum(r["input_tokens"] + r["cache_read_tokens"] for r in records)
        total_cache = sum(r["cache_read_tokens"] for r in records)
        total_cost = sum(r["cost_usd"] for r in records)
        total_savings = sum(r["cache_savings_usd"] for r in records)

        return {
            "period_days": days,
            "total_calls": len(records),
            "total_tokens": total_input,
            "cache_hit_ratio": round(total_cache / total_input, 4) if total_input else 0,
            "total_cost_usd": round(total_cost, 4),
            "total_savings_usd": round(total_savings, 4),
            "effective_discount_pct": round(total_savings / (total_cost + total_savings) * 100, 1) if total_cost + total_savings > 0 else 0,
            "avg_cost_per_call_usd": round(total_cost / len(records), 6),
        }

    def _load_recent(self, days: int) -> List[dict]:
        cutoff = datetime.utcnow().timestamp() - (days * 86400)
        results = []
        if not os.path.exists(self.storage_path):
            return results
        with open(self.storage_path) as f:
            for line in f:
                try:
                    r = json.loads(line)
                    if datetime.fromisoformat(r["timestamp"]).timestamp() >= cutoff:
                        results.append(r)
                except:
                    continue
        return results


# Global Singleton — use anywhere
observability = ClaudeObservability()
```

```python
"""
🧠 Claude Integration — Skill Registry Bridge + Knowledge Auto-Storage
Path: skills/claude-integration.py
Features: Universal API · Observability · Auto-Save Knowledge
"""

from typing import List, Dict, Any, Optional
from app.claude_client import ClaudeClient
from app.claude_observability import observability
from skills.create_knowledge_artifact import KnowledgeArtifactEngine
import time

class ClaudeSkill:
    def __init__(self):
        self.client = ClaudeClient()
        self.knowledge = KnowledgeArtifactEngine()
        self.default_model = "claude-sonnet-4-6"
        self.default_system = "You are a helpful, precise technical assistant. Answer concisely."

    async def ask(
        self,
        prompt: str,
        system: Optional[str] = None,
        model: Optional[str] = None,
        temperature: float = 0.0,
        max_tokens: int = 1024,
        save_as_knowledge: bool = True,
        knowledge_type: str = "knowledge-note"
    ) -> Dict[str, Any]:
        """
        Send question → track cost → auto-save as Knowledge Artifact
        
        Args:
            save_as_knowledge: Store response in Knowledge Registry (default: True)
        """
        start = time.perf_counter()
        response = await self.client.send_message(
            messages=[{"role": "user", "content": prompt}],
            model=model or self.default_model,
            system=system or self.default_system,
            temperature=temperature,
            max_tokens=max_tokens,
            cache_system_prompt=True
        )
        latency_ms = int((time.perf_counter() - start) * 1000)
        text = self.client.extract_text(response)

        # 📊 Track cost & usage
        model_used = response.get("model", model or self.default_model)
        usage = response.get("usage", {})
        obs = observability.record(model_used, usage, latency_ms)

        # 🧠 Auto-save to Knowledge Artifact Engine
        artifact_id = None
        if save_as_knowledge and len(text) > 20:
            artifact = await self.knowledge.create({
                "type": knowledge_type,
                "title": prompt[:80] + ("..." if len(prompt) > 80 else ""),
                "content": text,
                "source": {"type": "claude", "model": model_used},
                "confidence": 0.95,
                "related_skills": ["claude", "create-knowledge-artifact"]
            })
            artifact_id = artifact.get("artifact_id")

        return {
            "text": text,
            "model": model_used,
            "usage": usage,
            "cost": obs["cost"],
            "latency_ms": latency_ms,
            "cache_hit_ratio": obs["record"]["cache_hit_ratio"],
            "artifact_id": artifact_id,  # 🔗 Saved knowledge reference
            "message_id": response.get("id"),
        }

    async def summarize(self, text: str, **kwargs) -> Dict[str, Any]:
        """Summarize → auto-save as research-report"""
        return await self.ask(
            prompt=f"Summarize this:\n\n{text}",
            system="Concise summary.",
            knowledge_type="research-report",
            **kwargs
        )

    async def extract_knowledge(self, text: str) -> Dict[str, Any]:
        """Structured extraction → auto-save"""
        return await self.ask(
            prompt=f"Extract JSON: concepts, facts, procedures:\n\n{text}",
            system="Valid JSON only.",
            knowledge_type="technical-guide",
            temperature=0.1
        )


# Register in Skill Registry
from app.core.skill_registry import SkillRegistry
SkillRegistry._skills["claude"] = ClaudeSkill
```
