Payments Platform - Umbrella Helm Chart & GitHub Actions CI/CD

Umbrella Helm Chart Overview

Umbrella chart ที่รวม api, skip-payment-cronjob, และ postgres (DB).

Chart.yaml (umbrella)

apiVersion: v2
name: payments-platform
description: Umbrella chart for API, CronJob, and Postgres
type: application
version: 0.1.0
appVersion: "1.0.0"
dependencies:
  - name: api
    version: 0.1.0
    repository: "file://charts/api"
  - name: skip-payment
    version: 0.1.0
    repository: "file://charts/skip-payment"
  - name: postgres
    version: 10.3.17
    repository: "https://charts.bitnami.com/bitnami"

values.yaml (umbrella level)

global:
  imagePullSecrets: []
  env:
    LOG_LEVEL: info

api:
  image:
    repository: your-registry/api
    tag: v1.2.3
  replicaCount: 2
  service:
    type: ClusterIP
    port: 8080
  env:
    DATABASE_URL: "postgresql://postgres:password@payments-postgres:5432/payments"

skip-payment:
  image:
    repository: your-registry/skip-payment
    tag: latest
  cron:
    schedule: "0 0 * * *"
  env:
    DB_URL: "postgresql://postgres:password@payments-postgres:5432/payments"
    SUPABASE_KEY: "REPLACE_ME"

postgres:
  fullnameOverride: payments-postgres
  auth:
    username: postgres
    password: password
    database: payments
  primary:
    persistence:
      enabled: true
      size: 8Gi

GitHub Actions CI/CD Pipeline

.github/workflows/deploy.yml

name: CI/CD Deploy Payments Platform

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Helm
        uses: azure/setup-helm@v4
        with:
          version: v3.14.0

      - name: Setup Kubectl
        uses: azure/setup-kubectl@v4
        with:
          version: v1.30.0

      - name: Configure Kubeconfig
        run: |
          echo "${{ secrets.KUBECONFIG_CONTENT }}" > kubeconfig.yaml
          export KUBECONFIG=$PWD/kubeconfig.yaml

      - name: Build API image
        run: |
          docker build -t ${{ secrets.REGISTRY }}/api:${{ github.sha }} ./api
          echo "${{ secrets.REGISTRY_PASSWORD }}" | docker login ${{ secrets.REGISTRY }} -u ${{ secrets.REGISTRY_USER }} --password-stdin
          docker push ${{ secrets.REGISTRY }}/api:${{ github.sha }}

      - name: Build Skip-Payment image
        run: |
          docker build -t ${{ secrets.REGISTRY }}/skip-payment:${{ github.sha }} ./skip-payment
          docker push ${{ secrets.REGISTRY }}/skip-payment:${{ github.sha }}

      - name: Helm dependency update
        run: |
          helm dependency update ./payments-helm-chart

      - name: Helm upgrade
        run: |
          helm upgrade --install payments-platform ./payments-helm-chart \
            -f ./payments-helm-chart/values.yaml \
            --set api.image.tag=${{ github.sha }} \
            --set skip-payment.image.tag=${{ github.sha }}

Staging Workflow Example

name: CI/CD Deploy Payments Platform (staging)

on:
  push:
    branches:
      - develop
  workflow_dispatch:

jobs:
  build-and-deploy-staging:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Helm
        uses: azure/setup-helm@v4
        with:
          version: v3.14.0

      - name: Setup Kubectl
        uses: azure/setup-kubectl@v4
        with:
          version: v1.30.0

      - name: Configure Kubeconfig (staging)
        run: |
          echo "${{ secrets.KUBECONFIG_STAGING }}" > kubeconfig.staging.yaml
          export KUBECONFIG=$PWD/kubeconfig.staging.yaml

      - name: Login to registry
        run: |
          echo "${{ secrets.REGISTRY_PASSWORD }}" | docker login ${{ secrets.REGISTRY }} -u ${{ secrets.REGISTRY_USER }} --password-stdin

      - name: Build and push API image (staging tag)
        run: |
          docker build -t ${{ secrets.REGISTRY }}/api:staging-${{ github.sha }} ./api
          docker push ${{ secrets.REGISTRY }}/api:staging-${{ github.sha }}

      - name: Build and push Skip-Payment image (staging tag)
        run: |
          docker build -t ${{ secrets.REGISTRY }}/skip-payment:staging-${{ github.sha }} ./skip-payment
          docker push ${{ secrets.REGISTRY }}/skip-payment:staging-${{ github.sha }}

      - name: Helm dependency update
        run: |
          helm dependency update ./payments-helm-chart

      - name: Helm upgrade/install (staging)
        run: |
          helm upgrade --install payments-platform-staging ./payments-helm-chart \
            -n payments-staging --create-namespace \
            -f ./payments-helm-chart/values-staging.yaml \
            --set api.image.tag=staging-${{ github.sha }} \
            --set skip-payment.image.tag=staging-${{ github.sha }}
