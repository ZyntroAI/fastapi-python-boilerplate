/**
 * Configuration payload for compiling a Skill into concrete artifacts.
 */
export interface SkillInputConfig {
  /** The operational mode (e.g., 'auto-experiment', 'opponent', 'curriculum') */
  mode: 'standard' | 'auto-experiment' | 'opponent' | 'curriculum';
  /** Base archetype guiding the behavior pattern */
  archetype: string;
  /** Domain target (e.g., 'Productivity', 'Dev Tools', 'Education') */
  domain: 'Productivity' | 'Education' | 'Fitness' | 'Dev tools' | 'Marketing';
  /** Core mechanical lever (e.g., 'Timeboxing', 'Spaced Repetition', 'Falsifiable Test') */
  mechanic: string;
  /** Active constraints to be evaluated (e.g., Rhyme mode, Word limits, Output formats) */
  constraints: SkillConstraint[];
}

/**
 * Constraint rule definition evaluated by the compiler.
 */
export interface SkillConstraint {
  id: string;
  type: 'regex' | 'length' | 'rhyme' | 'custom_eval';
  value: string | number;
  weight: number; // 0.0 - 1.0 weight for SCAMPER scoring
  description: string;
}

/**
 * Compiled production artifacts output by the pipeline.
 */
export interface CompiledArtifacts {
  /** Human-readable skill specification */
  skillSpecMd: string;
  /** LLM-Free rule-based rubric */
  rubricMd: string;
  /** UI stub generator code */
  generateUIStubPy: string;
  /** Unit test skeleton */
  unitTestSkeletonPy: string;
  /** Explainable-by-Construction Trace Log */
  traceLog: string[];
}

3. Production-Ready React Hook: useSkillCompiler
React Custom Hook สำหรับบริหารจัดการ State การส่งผ่าน Pipeline, การตรวจสอบ Constraint และการเรนเดอร์ Artifacts
import { useState, useCallback } from 'react';

export interface UseSkillCompilerReturn {
  config: SkillInputConfig;
  artifacts: CompiledArtifacts | null;
  isCompiling: boolean;
  error: string | null;
  updateConfig: <K extends keyof SkillInputConfig>(key: K, value: SkillInputConfig[K]) => void;
  addConstraint: (constraint: SkillConstraint) => void;
  removeConstraint: (id: string) => void;
  compile: () => Promise<CompiledArtifacts>;
}

const DEFAULT_CONFIG: SkillInputConfig = {
  mode: 'standard',
  archetype: 'Falsifiable Habit Builder',
  domain: 'Productivity',
  mechanic: 'SCAMPER Constraint Verification',
  constraints: []
};

/**
 * React Hook to manage Skill-to-Artifact state and compilation.
 */
export function useSkillCompiler(initialConfig?: Partial<SkillInputConfig>): UseSkillCompilerReturn {
  const [config, setConfig] = useState<SkillInputConfig>({ ...DEFAULT_CONFIG, ...initialConfig });
  const [artifacts, setArtifacts] = useState<CompiledArtifacts | null>(null);
  const [isCompiling, setIsCompiling] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);

  const updateConfig = useCallback(<K extends keyof SkillInputConfig>(key: K, value: SkillInputConfig[K]) => {
    setConfig((prev) => ({ ...prev, [key]: value }));
  }, []);

  const addConstraint = useCallback((constraint: SkillConstraint) => {
    setConfig((prev) => ({
      ...prev,
      constraints: [...prev.constraints.filter(c => c.id !== constraint.id), constraint]
    }));
  }, []);

  const removeConstraint = useCallback((id: string) => {
    setConfig((prev) => ({
      ...prev,
      constraints: prev.constraints.filter(c => c.id !== id)
    }));
  }, []);

  /**
   * Runs LLM-Free rule compilation local logic and generates the 4 core artifacts.
   */
  const compile = useCallback(async (): Promise<CompiledArtifacts> => {
    setIsCompiling(true);
    setError(null);

    try {
      // Step 1: Explainable Trace Creation
      const traceLog = [
        `[Archetype]: ${config.archetype}`,
        `[Domain]: ${config.domain}`,
        `[Mechanic]: ${config.mechanic}`,
        `[Constraints]: ${config.constraints.map(c => c.id).join(', ') || 'None'}`,
        `[Proof Trace]: Generated deterministic rubric without LLM dependency.`
      ];

      // Step 2: Render skill_spec.md
      const skillSpecMd = `# Skill Spec: ${config.archetype} (${config.domain})
## Core Mechanic
${config.mechanic}

## Constraints & Rules
${config.constraints.map(c => `- **${c.id}**: ${c.description} (Weight: ${c.weight})`).join('\n') || 'No constraints set.'}

## Trace
${traceLog.map(t => `> ${t}`).join('\n')}
`;

      // Step 3: Render LLM-Free Rubric
      const rubricMd = `# Deterministic Local Rubric Matrix
| Rule ID | Type | Weight | Formula / Target |
| :--- | :--- | :--- | :--- |
${config.constraints.map(c => `| ${c.id} | ${c.type} | ${c.weight} | \`${c.value}\` |`).join('\n') || '| Default | Presence | 1.0 | Non-empty output |'}
`;

      // Step 4: Render generate_UI_stub.py
      const generateUIStubPy = `import streamlit as st

def render_ui():
    st.title("${config.archetype} - ${config.domain}")
    st.caption("Mechanic: ${config.mechanic}")
    user_input = st.text_area("Skill Action Input:")
    if st.button("Evaluate Constraints"):
        st.success("Execution complete")

if __name__ == "__main__":
    render_ui()
`;

      // Step 5: Render unit_test_skeleton.py
      const unitTestSkeletonPy = `import pytest

def test_constraint_validation():
    # Auto-generated unit test skeleton for ${config.domain}
    constraints = ${JSON.stringify(config.constraints)}
    assert len(constraints) == ${config.constraints.length}
`;

      const result: CompiledArtifacts = {
        skillSpecMd,
        rubricMd,
        generateUIStubPy,
        unitTestSkeletonPy,
        traceLog
      };

      setArtifacts(result);
      return result;
    } catch (err: any) {
      const errMsg = err.message || 'Compilation failed';
      setError(errMsg);
      throw new Error(errMsg);
    } finally {
      setIsCompiling(false);
    }
  }, [config]);

  return {
    config,
    artifacts,
    isCompiling,
    error,
    updateConfig,
    addConstraint,
    removeConstraint,
    compile
  };
}
