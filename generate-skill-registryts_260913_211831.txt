// generate-skill-registry.ts
// Node/TS script to auto-generate index-full.tsx from JSON skill indexes

import fs from "fs";
import path from "path";

interface Skill {
id: string;
name: string;
category: string;
path: string;
version?: string;
}

interface SkillRegistry {
version: string;
totalSkills: number;
categories: string[];
skills: Skill[];
}

// 🔑 Define categories and their directories
const categories: Record<string, string> = {
code: "packages/skills/code",
data: "packages/skills/data",
security: "packages/skills/security",
content: "packages/skills/content",
system: "packages/skills/system",
business: "packages/skills/business",
"ai-agent": "packages/skills/ai-agent",
integration: "packages/skills/integration"
};

// 📂 Collect skills from each category
function collectSkills(): Skill[] {
const skills: Skill[] = [];

for (const [category, dir] of Object.entries(categories)) {
const files = fs.readdirSync(dir, { withFileTypes: true });

files.forEach(file => {
if (file.isDirectory()) {
const skillPath = path.join(dir, file.name, "skill.json");
const skillPyPath = path.join(dir, file.name, "skill.py");

if (fs.existsSync(skillPath)) {
const skillData = JSON.parse(fs.readFileSync(skillPath, "utf-8"));
skills.push({
id: skillData.id || file.name,
name: skillData.name || file.name,
category,
path: skillPath,
version: skillData.version
});
} else if (fs.existsSync(skillPyPath)) {
skills.push({
id: file.name,
name: file.name.replace(/-/g, " "),
category,
path: skillPyPath
});
}
}
});
}

return skills;
}

// 🖥️ Generate TSX file
function generateTSX(registry: SkillRegistry) {
const tsxContent = `
// index-full.tsx
// Auto-generated by generate-skill-registry.ts

import React from "react";

interface Skill {
id: string;
name: string;
category: string;
path: string;
version?: string;
}

interface SkillRegistry {
version: string;
totalSkills: number;
categories: string[];
skills: Skill[];
}

const skillRegistry: SkillRegistry = $`{JSON.stringify(registry, null, 2)};

const SkillIndex: React.FC = () => (
<div>
<h1>ZyntroAI MCP Skill Registry</h1>
<p>Total Skills: {skillRegistry.totalSkills}</p>
{skillRegistry.categories.map(cat => (
<section key={cat}>
<h2>{cat}</h2>
<ul>
{skillRegistry.skills
.filter(skill => skill.category === cat)
.map(skill => (
<li key={skill.id}>
<strong>{skill.name}</strong> — <code>{skill.path}</code>
</li>
))}
</ul>
</section>
))}
</div>
);

export default SkillIndex;
`;

fs.writeFileSync("packages/skills/index-full.tsx", tsxContent, "utf-8");
}

// 🚀 Run generator
const skills = collectSkills();
const registry: SkillRegistry = {
version: "1.2",
totalSkills: skills.length,
categories: Object.keys(categories),
skills
};

generateTSX(registry);

console.log(✅ Generated index-full.tsx with${skills.length} skills`);
