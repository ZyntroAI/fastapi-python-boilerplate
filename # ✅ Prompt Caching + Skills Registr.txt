```python
"""
🤖 Claude REST API Client — Full Production Version
Path: app/claude_client.py
Features: Chat · Streaming · Tools · Prompt Caching · Observability
"""

import os
import json
import httpx
from typing import List, Dict, Any, Optional, AsyncGenerator, Union
from dotenv import load_dotenv

load_dotenv()

class ClaudeClient:
    def __init__(
        self,
        api_key: Optional[str] = None,
        base_url: str = "https://api.anthropic.com",
        api_version: str = "2026-01-01",
        timeout: int = 120
    ):
        self.api_key = api_key or os.getenv("ANTHROPIC_API_KEY")
        if not self.api_key:
            raise ValueError("ANTHROPIC_API_KEY not set")
        
        self.base_url = base_url.rstrip("/")
        self.api_version = api_version
        self.timeout = timeout
        self.headers = {
            "x-api-key": self.api_key,
            "anthropic-version": api_version,
            "content-type": "application/json"
        }

    async def send_message(
        self,
        messages: List[Dict[str, str]],
        model: str = "claude-sonnet-4-6",
        max_tokens: int = 1024,
        system: Optional[Union[str, List[Dict]]] = None,
        temperature: float = 0.0,
        top_p: float = 1.0,
        top_k: int = 40,
        stop_sequences: Optional[List[str]] = None,
        stream: bool = False,
        tools: Optional[List[Dict]] = None,
        tool_choice: Optional[Union[str, Dict]] = "auto",
        cache_system_prompt: bool = True
    ) -> Dict[str, Any]:
        """
        Send message to Claude API.
        
        Args:
            cache_system_prompt: Mark system prompt as cacheable (~90% cost reduction on repeats)
        """
        payload = {
            "model": model,
            "max_tokens": max_tokens,
            "messages": messages,
            "temperature": temperature,
            "top_p": top_p,
            "top_k": top_k,
            "stream": stream
        }

        if stop_sequences:
            payload["stop_sequences"] = stop_sequences

        # ✅ Prompt Caching Support
        if system:
            if cache_system_prompt and isinstance(system, str):
                payload["system"] = [{
                    "type": "text",
                    "text": system,
                    "cache_control": {"type": "ephemeral"}
                }]
            else:
                payload["system"] = system

        if tools:
            payload["tools"] = tools
            payload["tool_choice"] = tool_choice

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            resp = await client.post(
                f"{self.base_url}/v1/messages",
                headers=self.headers,
                json=payload
            )
            
            if resp.status_code != 200:
                raise Exception(f"API Error {resp.status_code}: {resp.text}")
            
            return resp.json()

    async def stream_message(
        self,
        messages: List[Dict[str, str]],
        model: str = "claude-sonnet-4-6",
        max_tokens: int = 1024,
        system: Optional[str] = None,
        temperature: float = 0.0,
        cache_system_prompt: bool = True
    ) -> AsyncGenerator[str, None]:
        """Stream response token-by-token"""
        
        system_payload = None
        if system:
            if cache_system_prompt:
                system_payload = [{
                    "type": "text",
                    "text": system,
                    "cache_control": {"type": "ephemeral"}
                }]
            else:
                system_payload = system

        payload = {
            "model": model,
            "max_tokens": max_tokens,
            "messages": messages,
            "system": system_payload,
            "temperature": temperature,
            "stream": True
        }

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            async with client.stream(
                "POST",
                f"{self.base_url}/v1/messages",
                headers=self.headers,
                json=payload
            ) as resp:
                async for line in resp.aiter_lines():
                    if line.startswith("data: "):
                        chunk = line[6:]
                        if chunk == "[DONE]":
                            break
                        try:
                            data = json.loads(chunk)
                            delta = data.get("delta", {})
                            if "text" in delta:
                                yield delta["text"]
                        except:
                            pass

    def extract_text(self, response: Dict) -> str:
        """Helper: Get text content from API response"""
        return "".join([
            block["text"]
            for block in response.get("content", [])
            if block.get("type") == "text"
        ])

    def get_cache_stats(self, response: Dict) -> Dict[str, int]:
        """Extract cache metrics from response headers/usage"""
        usage = response.get("usage", {})
        return {
            "input_tokens": usage.get("input_tokens", 0),
            "cache_read_tokens": usage.get("cache_read_input_tokens", 0),
            "cache_created_tokens": usage.get("cache_creation_input_tokens", 0),
            "output_tokens": usage.get("output_tokens", 0),
            "cache_hit_ratio": round(
                usage.get("cache_read_input_tokens", 0) / 
                max(1, usage.get("input_tokens", 0) + usage.get("cache_read_input_tokens", 0)),
                2
            )
        }
```

## 📄 New File: `skills/claude-integration.py` — Registry Bridge

```python
"""
🧠 Claude Integration — Skill Registry Bridge
Path: skills/claude-integration.py
Purpose: Make Claude API available to ALL skills via unified interface
"""

from typing import List, Dict, Any, Optional
from app.claude_client import ClaudeClient

class ClaudeSkill:
    """
    Universal Claude Skill — callable from any skill in the ecosystem.
    
    Usage from other skills:
        claude = ClaudeSkill()
        resp = await claude.ask("Explain this code...")
    """

    def __init__(self):
        self.client = ClaudeClient()
        self.default_model = "claude-sonnet-4-6"
        self.default_system = "You are a helpful, precise technical assistant. Answer concisely."

    async def ask(
        self,
        prompt: str,
        system: Optional[str] = None,
        model: Optional[str] = None,
        temperature: float = 0.0,
        max_tokens: int = 1024
    ) -> Dict[str, Any]:
        """Simple one-off question"""
        response = await self.client.send_message(
            messages=[{"role": "user", "content": prompt}],
            model=model or self.default_model,
            system=system or self.default_system,
            temperature=temperature,
            max_tokens=max_tokens,
            cache_system_prompt=True
        )
        return {
            "text": self.client.extract_text(response),
            "model": response.get("model"),
            "usage": response.get("usage"),
            "cache": self.client.get_cache_stats(response)
        }

    async def chat(
        self,
        messages: List[Dict[str, str]],
        system: Optional[str] = None,
        tools: Optional[List[Dict]] = None
    ) -> Dict[str, Any]:
        """Multi-turn conversation with optional tools"""
        response = await self.client.send_message(
            messages=messages,
            system=system or self.default_system,
            tools=tools,
            cache_system_prompt=True
        )
        return {
            "text": self.client.extract_text(response),
            "usage": response.get("usage"),
            "cache": self.client.get_cache_stats(response)
        }

    async def summarize(self, text: str, max_words: int = 100) -> str:
        """Quick summary — cached system prompt"""
        resp = await self.ask(
            prompt=f"Summarize in {max_words} words or less:\n\n{text}",
            system="You are a precise summarizer. Keep it concise."
        )
        return resp["text"]

    async def extract_knowledge(self, text: str) -> Dict[str, Any]:
        """Used by create-knowledge-artifact skill"""
        resp = await self.ask(
            prompt=f"""Extract structured knowledge from this text.
Return JSON with: concepts, facts, procedures, decisions, examples.

Text: {text}""",
            system="Output valid JSON only. No explanation."
        )
        return resp["text"]


# Register in Skill Registry
from app.core.skill_registry import SkillRegistry

SkillRegistry._skills["claude"] = ClaudeSkill
```
